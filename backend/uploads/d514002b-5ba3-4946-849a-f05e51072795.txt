export interface Property {
  id: string;
  titleKey: string;          // Translation key for i18n support
  descriptionKey: string;    // Translation key for description
  price: number;
  currency: string;
  type: 'apartment' | 'house' | 'villa' | 'commercial';
  status: 'for-sale' | 'for-rent';
  beds: number;
  baths: number;
  area: number;              // Size in square feet
  location: {
    city: string;
    neighborhood: string;
    coordinates: { lat: number; lng: number };
  };
  mainImage: string;         // References the public/assets/properties folder
  images: string[];
  amenities: string[];       // Tags for specifications matrix
  isFeatured: boolean;
  agentId: string;
}
